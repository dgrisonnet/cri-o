1Types

Everything that gets published is immutable. Observations carry their own token, monotonic time and field availability.

type StatsServer struct {
    collectionPeriod  Duration          // 0 = on demand, >0 = background sweeps
    statsAdmit        admission         // R: stats get independent admission
    metricsAdmit      admission         // both bound sweeps AND on-demand / cache-miss reads
    statsSweep        *sweeper          // one per view, see §3
    metricsSweep      *sweeper
    store             *resultStore
}

type resultStore struct {
    mu         Mutex                        // R: short lock, map ops only, never held across I/O
    live       map[ID]*lifecycleToken
    pods       map[ID]*podStatsEntry
    containers map[ID]*containerHistory    // shared by nested and standalone stats
    metrics    map[ID]*podMetricsEntry
}

type lifecycleToken struct{ id ID }       // identity = the pointer itself; no generation needed

type observation[V any] struct {
    token      *lifecycleToken
    mono       MonotonicTime             // R: order samples by in-process monotonic time
    wall       Time                      // R: preserve wall-clock timestamps for the wire
    source     counterSource             // kind + cgroup version + producer, see §7
    present    fieldSet                  // R: missing fields stay absent, not zero
    values     V
}

type containerHistory struct { latest, baseline *observation[containerValues] }
type podStatsEntry    struct { obs *observation[podCounters] }       // pod-level only
type podMetricsEntry  struct { obs *observation[metricValues] }

2Lifecycle tokens

Capture an identity token tied to the existing sandbox or container lifecycle before collecting. Invalidate it and evict its entries in the removal path.

// Called from the existing lifecycle: AddSandbox / AddContainer.
// Registering at creation (not lazily on first read) is what stops a stale
// caller from recreating an entry for an object that is already gone.
func (s *resultStore) register(id ID) {
    s.mu.Lock(); defer s.mu.Unlock()
    s.live[id] = &lifecycleToken{id: id}
}

// Called before any I/O. nil means "already removed: don't collect".
func (s *resultStore) token(id ID) *lifecycleToken {
    s.mu.Lock(); defer s.mu.Unlock()
    return s.live[id]
}

// Called from RemoveContainer / RemoveSandbox. Never waits for collection.
func (s *resultStore) invalidate(id ID) {
    s.mu.Lock(); defer s.mu.Unlock()
    delete(s.live, id)                       // R: invalidate the token ...
    delete(s.pods, id); delete(s.containers, id); delete(s.metrics, id)  // R: ... and evict its entries
}

3Periodic sweeps (collection_period > 0)

Give stats and metrics independent refresh paths … let stats publish while a metrics refresh is blocked. Allow at most one periodic sweep per view.

type sweeper struct {
    refresh  func(Context, *Sandbox)   // refreshPodStats | refreshPodMetrics
    workers  semaphore                   // sweep parallelism: admission capacity - reservedForRequests
    mu       Mutex
    inflight map[ID]bool                // pods of THIS view with a refresh still running
    dispatching atomic.Bool            // one dispatch loop per view (it can wait for a worker)
}

func (ss *StatsServer) run(ctx Context) {          // ctx is cancelled on Shutdown
    if ss.collectionPeriod == 0 { return }   // R: keep zero-period on-demand mode
    tick := NewTicker(ss.collectionPeriod)
    for {
        select {
        case <-ctx.Done(): return
        case <-tick.C:
            go ss.statsSweep.dispatch(ctx, ss.ListSandboxes())     // R: independent refresh paths:
            go ss.metricsSweep.dispatch(ctx, ss.ListSandboxes())   //    a blocked metrics sweep never delays stats
        }
    }
}

// Fan out the pods of one tick across a bounded set of workers.
func (sw *sweeper) dispatch(ctx Context, sboxes []*Sandbox) {   // snapshot; no stateLock held
    if !sw.dispatching.CompareAndSwap(false, true) { return }  // last tick's loop still waiting for workers
    defer sw.dispatching.Store(false)                          // (all workers busy): don't stack loops
    for _, sb := range sboxes {
        if !sw.claim(sb.ID) { continue }      // still refreshing from an earlier tick: skip this pod only

        if err := sw.workers.Acquire(ctx); err != nil {   // acquire BEFORE spawning: goroutines bounded too
            sw.unclaim(sb.ID); return             // shutting down
        }
        go func(sb *Sandbox) {
            defer sw.workers.Release()
            defer sw.unclaim(sb.ID)
            sw.refresh(ctx, sb)                  // takes a slot from the view's admission budget (§4, §5)
        }(sb)
    }
    // No wait: dispatch returns once every eligible pod has a worker.
    // Slow pods keep running and are simply skipped on the next tick.
}

func (sw *sweeper) claim(id ID) bool {
    sw.mu.Lock(); defer sw.mu.Unlock()
    if sw.inflight[id] { return false }
    sw.inflight[id] = true
    return true
}
Decision: parallel, bounded. Each tick fans pods out to at most workers goroutines per view. Two limits apply:
  • Sweep workers (sweeper.workers) cap how many pods one view refreshes at once. It's sized below the view's admission capacity, so a full sweep always leaves reservedForRequests slots for cache misses and on-demand RPCs.
  • Admission (statsAdmit / metricsAdmit) caps all concurrent reads in that view, sweeps and requests together. That's the review's "bound concurrent reads, including synchronous cache misses and on-demand requests".
Why in-flight is tracked per pod, not per sweep. The review says at most one periodic sweep per view. Read literally (one flag per view), a single pod stuck in an uninterruptible driver call would hold the whole sweep open, every later tick would be skipped, and no pod in that view would refresh. Tracking in-flight per pod keeps the intent (work never piles up: at most one refresh per pod per view) while the other pods keep refreshing every period. A stuck pod costs one worker, not the view.
OPEN. The values of workers and reservedForRequests per view. Metrics probably wants fewer workers than stats because of disk I/O. The review asks to measure read concurrency with disk metrics enabled before picking (implementation step 3).

4Stats collection

Each RPC selects its own inputs. Use one container stats builder for standalone and nested container stats. Periodic pod stats refresh should publish child stats to the same per-container history used by standalone RPCs.

func (ss *StatsServer) refreshPodStats(ctx Context, sb *Sandbox) {
    podTok := ss.store.token(sb.ID)           // R: capture token BEFORE collecting
    if podTok == nil { return }

    release, err := ss.statsAdmit.acquire(ctx)  // R: stats slots can't be filled by disk work
    if err != nil { return }
    defer release()

    // Pod-level counters: host pod cgroup. Process count from cgroup PID counters,
    // not a process scan.                                     R: pod stats already get this
    pod := ss.adapter.readPodCore(ctx, sb)     // -> observation[podCounters]
    pod.token = podTok
    ss.store.publishPod(pod)                  // rejected silently if pod was removed meanwhile

    for _, c := range sb.LiveContainers() {
        ss.collectContainerStats(ctx, c, sb)   // same builder as the standalone RPC
    }
}

// The ONE container stats builder: used by the pod sweep, by ListContainerStats on
// a miss, and by on-demand mode. Stats inputs only: no rootfs walk, no process scan.
func (ss *StatsServer) collectContainerStats(ctx Context, c *Container, sb *Sandbox) *observation[containerValues] {
    tok := ss.store.token(c.ID)
    if tok == nil { return nil }

    obs, err := ss.adapter.readContainerCore(ctx, c, sb)   // cgroup CPU/memory (or VM shim task call)
    if err != nil {
        return nil                               // R: required read failed: keep previous sample + age
    }
    if wl, err := ss.writableLayer(ctx, c); err == nil {   // CRI WritableLayer = storage-driver usage
        obs.values.writable = wl; obs.present.add(WritableLayer)
    }                                            // R: optional read failed: field stays absent
    obs.token = tok
    ss.store.publishContainer(obs)             // R: same history as standalone RPCs
    return obs
}

5Metrics collection

Collect optional metrics before rendering so projection operates on values. Separate the optional process scan from the core read and run it only for metrics that consume it.

func (ss *StatsServer) refreshPodMetrics(ctx Context, sb *Sandbox) {
    tok := ss.store.token(sb.ID)
    if tok == nil { return }

    release, err := ss.metricsAdmit.acquire(ctx)
    if err != nil { return }
    defer release()

    enabled := ss.EnabledPodMetrics()
    vals := metricValues{}
    obs := observation[metricValues]{token: tok, mono: MonoNow(), wall: Now()}

    for _, c := range sb.LiveContainers() {
        core, err := ss.adapter.readContainerCore(ctx, c, sb)   // reuse core read code; may read twice vs stats
        if err != nil { continue }
        vals.addCore(c, core)

        if enabled.needsProcessScan() {          // R: process scan only if a metric consumes it
            if p, err := ss.adapter.scanProcesses(ctx, c); err == nil { vals.addProcesses(c, p) }
        }
        if enabled.has(DiskMetrics) {
            if d, err := ss.adapter.diskUsage(ctx, c); err == nil { vals.addDisk(c, d) }
        }
    }
    if enabled.has(NetworkMetrics) { vals.addNetwork(ss.adapter.readNetwork(ctx, sb)) }

    obs.values = vals
    ss.store.publishMetrics(&obs)              // rendering to CRI metrics happens later, on values, no I/O
}

6Runtime adapter

Carry source and field availability through the adapter, and normalize the memory and swap fields both RPCs use before projection. Pass the request context to the shim task call itself.

func (a *adapter) readContainerCore(ctx Context, c *Container, sb *Sandbox) (*observation[containerValues], error) {
    obs := &observation[containerValues]{mono: MonoNow(), wall: Now()}
    switch c.Runtime {
    case VM:
        resp, err := shim.Stats(ctx, c.ID)    // R: request ctx reaches the task call
        if err != nil { return nil, err }
        raw, src := decodeShimMetrics(resp)       // version from the payload type, see §7
        obs.source = src                           // R: guest version, not host
        obs.values, obs.present = normalize(raw, src)  // VM disk fields: absent, not 0
    default:
        cg, err := cgroup.CoreStats(c.CgroupPath)   // R: no GetPids / fd scan in the core read
        if err != nil { return nil, err }
        src := counterSource{kind: hostContainerCgroup, version: hostCgroupVersion, producer: libcontainerFor(hostCgroupVersion)}
        obs.source = src
        obs.values, obs.present = normalize(rawFromLibcontainer(cg), src)   // see §7
    }
    return obs, nil
}

func (a *adapter) diskUsage(ctx Context, c *Container) (diskValues, error) {
    ref, ok := c.StableMountRef()             OPEN // R: "investigate" — may not exist for every runtime
    if !ok {
        return c.DiskStatsUnderOpLock(ctx)        // fallback: today's walk; stop dependency stays explicit
    }
    defer ref.Release()
    usage, err := boundedWalk(ctx, ref, maxEntries)   // R: bounded, cancellable, outside opLock
    if err != nil { return diskValues{}, err }
    capacity, inodes := statfs(ref)                // same scope as usage, as today
    return diskValues{usage, capacity, inodes}, nil
}

7Cgroup v1/v2 normalization

Host and guest counters need different interpretation. A VM shim response identifies the guest cgroup version. Current projections use the host version for memory keys and swap arithmetic. Normalization happens once in the adapter. Nothing after it calls node.CgroupIsV2().

// ---- Where a sample came from --------------------------------------------

type counterSource struct {
    kind     sourceKind   // hostPodCgroup | hostContainerCgroup | guestContainer
    version  int          // cgroup version of THIS sample (1 | 2), not of the node
    producer producer     // which library produced the raw numbers (decides swap semantics)
}

const (
    libcontainerFS   producer = iota   // host v1:  SwapUsage = memsw        (mem + swap)
    libcontainerFS2                   // host v2:  SwapUsage = swap.current + memory.current
    containerdV1                      // guest v1: Swap.Usage = memsw       (mem + swap)
    containerdV2                      // guest v2: SwapUsage  = swap only
)

var hostCgroupVersion = ternary(node.CgroupIsV2(), 2, 1)   // read once at startup; only used to label host samples

// The shim tells us the guest version through the payload type. No guessing from the host.
func decodeShimMetrics(resp *task.StatsResponse) (rawCgroup, counterSource) {
    switch m := typeurl.UnmarshalAny(resp.Stats).(type) {
    case *cgroupsV1.Metrics:
        return rawFromContainerdV1(m), counterSource{guestContainer, 1, containerdV1}
    case *cgroupsV2.Metrics:
        return rawFromContainerdV2(m), counterSource{guestContainer, 2, containerdV2}
    }
    // unknown payload: required read failed -> caller keeps previous sample
}

// ---- Version-neutral values: every field has ONE meaning on every source ---

type memoryValues struct {
    usage, workingSet, rss     uint64
    swap, swapLimit            uint64   // swap ONLY, everywhere
    pageFaults, majorFaults    uint64   // v2 only
    anonTHP, shmemTHP, fileTHP uint64   // v2 only
    activeAnon, inactiveAnon   uint64
    fileMapped                 uint64
}

func normalizeMemory(raw rawMemory, src counterSource) (memoryValues, fieldSet) {
    m := memoryValues{usage: raw.usage}
    present := fieldSet{Usage, WorkingSet, RSS, Swap, Anon, FileMapped}

    switch src.version {                     // R: branch on the SAMPLE's version, not the node's
    case 2:
        m.workingSet = satSub(raw.usage, raw.stat["inactive_file"])
        m.rss        = raw.stat["anon"]
        m.activeAnon, m.inactiveAnon = raw.stat["active_anon"], raw.stat["inactive_anon"]
        m.fileMapped = raw.stat["file_mapped"]
        m.pageFaults, m.majorFaults = raw.stat["pgfault"], raw.stat["pgmajfault"]
        m.anonTHP, m.shmemTHP, m.fileTHP = raw.stat["anon_thp"], raw.stat["shmem_thp"], raw.stat["file_thp"]
        present.add(PageFaults, THP)
    case 1:
        m.workingSet = satSub(raw.usage, raw.stat["total_inactive_file"])
        m.rss        = raw.stat["total_rss"]
        m.activeAnon, m.inactiveAnon = raw.stat["total_active_anon"], raw.stat["total_inactive_anon"]
        m.fileMapped = ternary(raw.useHierarchy, raw.stat["total_mapped_file"], raw.stat["mapped_file"])
        // v1 has no pgfault/pgmajfault in this path and no THP keys:
        // leave them ABSENT.                                     R: absent, not measured zero
    }
    for _, k := range raw.missingKeys(src.version) { present.remove(fieldFor(k)) }  // e.g. a decoder that never sets a key

    m.swap, m.swapLimit = swapOnly(raw, src.producer)
    return m, present
}

// The ONLY place that knows each producer's swap convention.
func swapOnly(raw rawMemory, p producer) (usage, limit uint64) {
    switch p {
    case libcontainerFS, libcontainerFS2, containerdV1:   // reported as mem + swap
        return satSub(raw.swapReported, raw.usage), satSub(raw.swapLimitReported, raw.usageLimit)
    case containerdV2:                                   // already swap only
        return raw.swapReported, raw.swapLimitReported
    }
}
What this fixes vs today. computeMemoryStats, computeSwapUsage and the metrics helpers branch on node.CgroupIsV2(). So a v2 guest on a v1 host reads total_* keys the v2 decoder never wrote (working set = usage, RSS = 0). A v1 guest on a v2 host loses anon. A v2 guest's swap-only value gets memory.current subtracted from it and usually reads 0. Here the version and producer travel with the sample.
To verify before relying on it. On a v1 host, today's computeSwapUsage returns libcontainer's memsw value as-is. That's memory + swap (vendor/github.com/opencontainers/cgroups/fs/memory.go:172), which suggests v1 hosts currently report mem+swap as swap. The swap limit subtraction for each producer also needs checking against kubelet's expectations. The review doesn't mention either.
Tests (review, implementation step 3). Each producer gets a fixture-driven table test: guest v1 on a v2 host, guest v2 on a v1 host, swap enabled/disabled/unlimited, and a key missing from the decoder → field absent, not 0. normalizeMemory takes counterSource as a parameter, so every branch runs on any CI host. computeAnonMemory(mem, isCgroupV2) already does this today.

8Publish

Under a short store lock, make token validation and publication atomic with respect to that invalidation. Reject an older observation if a newer one has already been accepted.

func (s *resultStore) publishContainer(obs *observation[containerValues]) bool {
    s.mu.Lock(); defer s.mu.Unlock()            // no I/O below this line

    if s.live[obs.token.id] != obs.token {     // R: removed during collection -> drop
        return false
    }
    h := s.containers[obs.token.id]
    if h == nil { h = &containerHistory{}; s.containers[obs.token.id] = h }

    if h.latest != nil && !newer(obs, h.latest) { // R: older observation loses
        return false
    }
    // Copy-on-write: never mutate a published observation; an RPC may be encoding it.
    h2 := &containerHistory{baseline: h.latest, latest: obs}   // R: immutable after publication
    if h.latest != nil && h.latest.source != obs.source { h2.baseline = nil }  // new counter source: new baseline
    s.containers[obs.token.id] = h2
    return true
}

func newer(a, b *observation[V]) bool {
    if a.mono != b.mono { return a.mono.After(b.mono) }
    return tieBreak(a, b)                        // R: deterministic tie rule  OPEN which one
}

9RPC read paths

Build pod responses from current membership and eligible child entries so invalidating a child hides it without rewriting a published pod response.

func (ss *StatsServer) StatsForContainers(ctx Context, ctrs []*Container) []*ContainerStats {
    out := []*ContainerStats{}
    for _, c := range ctrs {
        h := ss.store.readContainer(c.ID)        // lock: lookup + liveness check only
        if ss.collectionPeriod == 0 || h == nil { // on-demand mode, or cache miss
            ss.collectWithAdmission(ctx, c)     // statsAdmit; caller owns its own work (no coalescing yet)
            h = ss.store.readContainer(c.ID)
        }
        if h == nil { continue }                   // removed, or required read failed with no previous
        if tooOld(h.latest) { continue }          OPEN // R: agree on a maximum age first
        out = append(out, buildContainerStats(h))   // pure projection on values
    }
    return out
}

func (ss *StatsServer) StatsForSandboxes(ctx Context, sboxes []*Sandbox) []*PodSandboxStats {
    out := []*PodSandboxStats{}
    for _, sb := range sboxes {
        pod := ss.store.readPod(sb.ID)
        if ss.collectionPeriod == 0 || pod == nil { ss.refreshPodStats(ctx, sb); pod = ss.store.readPod(sb.ID) }
        if pod == nil { continue }
        children := []*containerHistory{}
        for _, c := range sb.LiveContainers() {   // R: current membership, at read time
            if h := ss.store.readContainer(c.ID); h != nil { children = append(children, h) }
        }
        out = append(out, buildPodStats(pod, children))
    }
    return out
}

10CPU rate

Calculate CPU rate only between ordered observations for the same identity and counter source with positive elapsed time. A counter reset starts a new baseline.

func cpuRate(h *containerHistory) (nanoCores uint64, ok bool) {
    cur, prev := h.latest, h.baseline
    if prev == nil                          { return 0, false }   // first sample: omit the field
    if cur.source != prev.source            { return 0, false }   // R: same counter source (kind, version, producer)
    elapsed := cur.mono.Sub(prev.mono)
    if elapsed <= 0                         { return 0, false }   // R: positive elapsed time
    if cur.values.cpuTotal < prev.values.cpuTotal { return 0, false } // R: reset -> new baseline (next sample)
    return (cur.values.cpuTotal - prev.values.cpuTotal) * 1e9 / uint64(elapsed), true
}
Because the pod sweep and standalone misses both write containerHistory through the same ordered publish, nested and standalone CPU rates now share one baseline. That fixes flaw #4 in current-design.md.
